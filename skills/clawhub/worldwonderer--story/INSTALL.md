clawhub install @worldwonderer/story-cover --version 1.1.18
clawhub install @worldwonderer/story-deslop --version 1.1.62
clawhub install @worldwonderer/story-import --version 1.0.55
clawhub install @worldwonderer/story-long-analyze --version 1.1.49

clawhub install @worldwonderer/story-short-analyze --version 1.1.38
