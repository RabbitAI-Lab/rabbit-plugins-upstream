clawhub install @worldwonderer/story-deslop --version 1.1.47
clawhub install @worldwonderer/story-import --version 1.0.43
clawhub install @worldwonderer/story-long-analyze --version 1.1.40

clawhub install @worldwonderer/story-short-analyze --version 1.1.31
