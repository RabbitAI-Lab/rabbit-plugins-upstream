# image-with-comfyui — SFW model download sources
#
# This file lists clean / official download sources for the model weights
# required by the image-with-comfyui skill. Everything here is SFW.
#
# Common prefix: https://huggingface.co/
# File path = "{prefix}/{repo}/{path}"
#
# Format legend:
#   ✓  = file/weight verified to exist
#   ★  = repo may require login (HTTP 401 without an authenticated HF account);
#        download on a machine where you are logged in.
#
# -----------------------------------------------------------------
# FILENAME CAVEAT (read before deploying)
# -----------------------------------------------------------------
# The SFW weights below usually have DIFFERENT filenames than what the bundled
# workflow's loader nodes expect. The script does NOT rename model files for
# you. If you deploy the SFW weights you must make the workflow match the
# on-disk filenames, EITHER by:
#   (a) renaming the downloaded files to match the workflow node's
#       unet_name / ckpt_name, OR
#   (b) editing that node's unet_name / ckpt_name to the actual filenames.
# For Qwen-Image 2.1 the loader node is node 17 in
#   workflows/qwen_image_2.1_api.json (unet_name).
# For **Qwen-Image 2.1 Turbo** (`qwen21t`) the loader node is node 17 in
#   workflows/qwen_image_2.1_turbo_api.json (unet_name).
# The turbo and legacy workflows have identical node topology; only the
# unet_name (filename) differs — pick the one that matches the file on disk.
# For SD3.5 the loader node is node 4 in
#   workflows/sd3.5-med_t2i_api.json (ckpt_name).
# For Wan2.2 the loader nodes are 307 (high) and 308 (low) in
#   workflows/wan2.2_i2v_api.json (unet_name).
# Leave the workflow's unet_name/ckpt_name pointing at the filenames that
# actually exist on disk after you place the weights.
# -----------------------------------------------------------------

# -----------------------------------------------------------------
[1] Qwen-Image 2.1  --  default T2I + I2I (single universal model)
# Expected location (as specified by workflows/qwen_image_2.1_api.json, node 17):
#   diffusion model   -> models/diffusion_models/<folder>/<filename>.safetensors
#   text encoder      -> models/text_encoders/
#   VAE               -> models/vae/
#
# (a) Official / clean (recommended) -- Comfy-Org sharded repo ✓
# https://huggingface.co/Comfy-Org/Qwen-Image-2.1/tree/main?recursive=true
#       diffusion_models/qwen_image_2.1_int8_convrot.safetensors   # official int8
#       # (place under models/diffusion_models/, keep the intended subfolder)
#     Text encoder -> models/text_encoders/
#       text_encoders/qwen3vl_8b_int8_convrot.safetensors        # Qwen3-VL 8B int8
#     VAE          -> models/vae/
#       vae/qwen_image_2.1_vae_bf16.safetensors                  # official VAE bf16
#
# Loading note: this skill loads standard safetensors via UNETLoader.
#   weight_dtype must be fp8_e4m3fn_fast; CFG must = 1.0 (distilled model).
#   The official SFW diffusion model ships under the diffusion_models/ subfolder.
#   Match the workflow node 17 unet_name to the exact filename/folder on disk.

# -----------------------------------------------------------------
[1b] Qwen-Image 2.1 **Turbo**  --  fast 8-step default (id `qwen21t`)
# Expected location (as specified by workflows/qwen_image_2.1_turbo_api.json,
# node 17):
#     diffusion model -> models/diffusion_models/QWEN_Image/
#                        qwen-image-2.1-turbo-int8_convrot.safetensors  # the
#                        exact file the turbo workflow expects (note the
#                        "turbo" + int8_convrot in the name)
#     text encoder    -> models/text_encoders/qwen3vl_8b_int8_convrot.safetensors
#     VAE             -> models/vae/qwen_image_2.1_vae_bf16.safetensors
#
# (a) Quantized builds for the SAME turbo base weights (official turbo):
#     Repo: abenzerps/Qwen-Image-2.1-Turbo-Quantized ✓ (SFW)
#     https://huggingface.co/abenzerps/Qwen-Image-2.1-Turbo-Quantized
#       qwen-image-2.1-turbo-int8_convrot.safetensors   # 6.76 GB — the file this
#       skill uses (matches the on-disk name above)
#       qwen-image-2.1-turbo-fp8.safetensors            # 6.63 GB
#       qwen-image-2.1-turbo-NVFP4.safetensors          # 4.20 GB
#       (GGUF / MLX 4-8 bit variants also available — do NOT use a non-int8
#        name unless you also edit node 17's unet_name to match it)
#     Companion text encoder + VAE are packaged in the repo too:
#       text_encoders/qwen3vl_8b_int8_convrot.safetensors   # 9.35 GB
#       vae/qwen_image_2.1_vae_bf16.safetensors             # 676 MB
#
# (b) Original (unquantized) turbo base weights:
#       Qwen/Qwen-Image-2.1-Turbo   (the base the quantizations above derive from)
#       Comfy-Org/Qwen-Image-2.1    (text encoder + VAE source)
#
# (c) Legacy Qwen-Image 2.1 (id `qwen21`, still supported): see item [1] above.
#
# Loading note: turbo loads via the SAME UNETLoader + weight_dtype
#   fp8_e4m3fn_fast + CFG 1.0 as legacy qwen21. The 8-step turbo schedule is
#   a fast distilled path; quality is in the same band as the 25-step legacy.
#   Match node 17 in qwen_image_2.1_turbo_api.json to the file you download.
