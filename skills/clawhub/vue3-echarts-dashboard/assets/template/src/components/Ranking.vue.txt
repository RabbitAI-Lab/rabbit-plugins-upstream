<template>
  <div class="ranking">
    <div v-for="(item, i) in list" :key="item.name" class="ranking__row">
      <span class="ranking__idx" :class="{ 'ranking__idx--top': i < 3 }">{{ i + 1 }}</span>
      <span class="ranking__name" :title="item.name">{{ item.name }}</span>
      <div class="ranking__bar-wrap">
        <div class="ranking__bar" :style="{ width: pct(item.value) + '%' }" />
      </div>
      <span class="ranking__val">{{ item.value.toLocaleString() }}</span>
    </div>
  </div>
</template>

<script setup lang="ts">
import { computed } from 'vue'
import type { PieItem } from '@/api/types'

const props = defineProps<{ data: { title?: string; list: PieItem[] } | null }>()

const list = computed(() => props.data?.list ?? [])
const max = computed(() => Math.max(1, ...list.value.map((d) => d.value)))

function pct(v: number) {
  return Math.round((v / max.value) * 100)
}
</script>
