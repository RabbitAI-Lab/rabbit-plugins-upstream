import cvxopt as op  
from cvxopt import matrix  
P = matrix([[0.048,0.0,0.0],[0.0,0.189,0.0],[0.0,0.0,0.139]])  
q = matrix([0.0,0.0,0.0])  
G = matrix([[1.,0.,0.,-1.,0.,0.,-0.179],[0.,1.,0.,0.,-1.,0.,-0.131],[0.,0.,1.,0.,0.,-1,-0.283]],(7,3))     #7X3阶矩阵  
A = matrix([1.0,1.0,1.0],(1,3))       #1x3阶矩阵  
b = matrix(1.0)  
h = matrix([0.5,0.5,0.5,-0.1,-0.1,-0.1,-0.15])  
sol = op.solvers.qp(P,q,G,h,A,b)  
print(sol['x'])
