def OLS(y_dataframe,x_dataframe):
	resultlist=pd.Series(index=x_dataframe.index)
	for i in range(0,len(y_dataframe.index)):
		X= sm.add_constant(x_dataframe.iloc[i])
		result= sm.OLS(y_dataframe.iloc[i],X).fit()
		index=x_dataframe.index[i]
		resultlist[index]= result.params[1]
	return resultlist
