			{'stock':'600028.SH','weight':0.11,'quantity':200,'optType':24},
		]
	basket={'name':'basket1','stocks':table}
	set_basket(basket)
	# 按篮子数量下单， 下2份 # 即下两倍篮子
	pice = 2

			'',2,'strReMark',C)
	
	# 按篮子权重下单

			{'stock':'600000.SH','weight':0.4,'quantity':0,'optType':23}, # 40%

		]

	set_basket(basket)
