#获取中性化后的因子
def OLS(DataFrame): 
	y=DataFrame['VOL20_M_Z']
	x=DataFrame.drop(['VOL20_M_Z'],axis=1)
	Y = y.fillna(0)
	X = x.fillna(0)
	LR = Lasso()
	LR.fit(X, Y)
	factor = pd.DataFrame(Y) - pd.DataFrame(LR.predict(X), index = Y.index, columns = ['VOL20_M_Z'])
	return factor
