from datetime import datetime
import backtrader as bt
import os.path  # 管理路径
import sys  # 发现脚本名字(in argv[0])
from backtrader.feeds import GenericCSVData  # 用于扩展DataFeed


# 创建新的data feed类
class GenericCSV_PE(GenericCSVData):

    # 增加pe线
    lines = ('m', )
    # 默认第8列（从0开始数）
    params = (('m', 13), )


# 创建策略类
class SmaCross(bt.Strategy):
    # 定义参数
    params = dict(period=5  # 移动平均期数
                  )

    # 日志函数
    def log(self, txt, dt=None):
        '''日志函数'''
        dt = dt or self.datas[0].datetime.date(0)
        print('%s, %s' % (dt.isoformat(), txt))

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            # 订单状态 submitted/accepted，无动作
            return

        # 订单完成
        if order.status in [order.Completed]:
            if order.isbuy():
                # self.log('买单执行, %.2f' % order.executed.price)
                self.log(f'买单执行,{order.executed.price}, {order.size}')

            elif order.issell():
                self.log(f'卖单执行,{order.executed.price}, {order.size}')

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.log('订单 Canceled/Margin/Rejected')

        self.log(f'value {self.broker.getvalue()}, position szie {self.position.size}')

    # 记录交易收益情况（可省略，默认不输出结果）
    def notify_trade(self, trade):
        if trade.isclosed:
            print('毛收益 %0.2f, 扣佣后收益 % 0.2f, 佣金 %.2f' %
                  (trade.pnl, trade.pnlcomm, trade.commission))

    def __init__(self):
        pass



    def next(self):
        self.log('订单创建，目标数量, %.2f' % self.data.m[0])
        m=self.data.m[0]
        if m>0:
            size=1
        elif m==0:
            size=0
        else:
            size=-1


        self.order_target_size(data=self.data,target=size)

# 创建大脑引擎对象
cerebro = bt.Cerebro()

# 获取本脚本文件所在路径
modpath = os.path.dirname(os.path.abspath(sys.argv[0]))
# 拼接得到数据文件全路径
# datapath = os.path.join(modpath, './600000pe.csv')
datapath = r'E:\\myquant\\Adv_Fin_ML\\research-master\\Advances in Financial Machine Learning\\Labelling\\official_data\\data_to_backtest.csv'
# 创建行情数据对象，加载数据
data = GenericCSV_PE(
        dataname=datapath,
        datetime=0,  # 日期行所在列
        open=2,  # 开盘价所在列
        high=3,  # 最高价所在列
        low=4,  # 最低价所在列
        close=5,  # 收盘价价所在列
        volume=6,  # 成交量所在列
        openinterest=-1,  # 无未平仓量列
        m = 13,
        dtformat=('%Y-%m-%d %H:%M:%S.%f'),  # 日期格式
        # fromdate=datetime(2020, 1, 1),  # 起始日

    )

cerebro.addstrategy(SmaCross)  # 将策略注入引擎

cerebro.broker.setcash(10000.0)  # 设置初始资金

print('初始市值: %.2f' % cerebro.broker.getvalue())
cerebro.run()  # 运行
