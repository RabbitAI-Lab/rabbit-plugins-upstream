# 默认的GenericCSVData只包含datetime,open,high,close,volime,openinterest7根数据
# 若需要增加一个类似pe。

from datetime import datetime
import backtrader as bt
import os.path
import sys
from backtrader.feeds import GenericCSVData


class GenericCSV_PE(GenericCSVData):
    line = ('pe',)  # 增加一列pe
    params = (('pe', 8),)  # 从第八列开始

class GenericCSV_PE(GenericCSVData):

    # 增加pe线
    lines = ('pe', )
    # 默认第8列（从0开始数）
    params = (('pe', 8), )

class SmaCross(bt.Strategy):
    params = dict(period=5)

    def log(self, txt, dt=None):
        dt = dt or self.datas[0].datetime.date(0)
        print('%s,%s' % (dt.isoformat(), txt))

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                self.log('买单执行，%.2f' % order.executed.price)
            elif order.issell():
                self.log('卖单执行， %.2f' % order.executed.price)
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            self.log('订单 Canceled/Margin/Rejected')

    # 记录交易情况
    def notify_trade(self, trade):
        if trade.isclosed:
            print('毛收益 %0.2f,扣佣后收益 %0.2f,佣金 %.f' %
                  (trade.pnl, trade.pnlcomm, trade.commission))

    def __init__(self):
        self.move_average = bt.ind.MovingAverageSimple(
            self.data, period=self.params.period
        )
        # 交叉信号
        self.crossover = bt.ind.CrossOver(self.data, self.move_average)

        print(self.data.lines.getlinealiases())

    def next(self):

        if not self.position:
            if self.crossover > 0 :
                self.log('创建买单')
                self.buy(size=100)
        elif self.crossover < 0:
            self.log('创建卖单')
            self.sell(size=100)


cerebro = bt.Cerebro()
modpath = os.path.dirname(os.path.abspath(sys.argv[0]))
datapath = os.path.join(modpath, 'CBT1.csv')

data = GenericCSV_PE(
    dataname=datapath,
    datetime=0,
    open=1,
    high=2,
    low=3,
    close=4,
    volume=5,  # 若不需要，则写-1
    openinterest=-1,  # 无未平仓量列
    pe=9,
    dtformat=('%Y/%m/%d'),
    fromdate=datetime(2020, 1, 2),
    todate=datetime(2020, 12, 31))

cerebro.adddata(data)
cerebro.addstrategy(SmaCross)
cerebro.broker.setcash(10000.0)
cerebro.broker.setcommission(0.001)
cerebro.broker.set_slippage_fixed(0.05)
print('初始市值： %.2f' % cerebro.broker.getvalue())
cerebro.run()
print('最终市值： %.2f' % cerebro.broker.getvalue())
cerebro.plot()
